reg [3:0] bin_counter; // Ваша 4-битная шина с делением частоты

initial begin
    bin_counter = 0;
    
    // Цикл выполнится 16 раз и покроет все комбинации
    repeat (16) begin
        #10; // Период изменения младшего бита (частота задается тут)
        bin_counter = bin_counter + 1;
    end
    
    $finish; // Конец симуляции
end
